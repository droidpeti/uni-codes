< U, Ipr, Ifn, Iconst> ahol

U = {i, j, k}
Ipr: P -> Pi : Pi ig = {(i,j), (j, k), (k,j)}
     Q -> Qi : Qi ig = {(j), (k)}

Ifn: f -> fi : Fi = j (konstans fv)
     g -> gi : gi(x) = x (identitás fv)
     h -> hi : hi: i -> k
                   j -> i
                   k -> j

Iconst: 1 -> i
        2 -> k
